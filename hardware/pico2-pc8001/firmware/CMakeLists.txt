cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2 CACHE STRING "Pico 2 RP2350 board")
if(NOT PICO_BOARD STREQUAL "pico2")
  message(FATAL_ERROR "This validation build targets Raspberry Pi Pico 2 only")
endif()
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/pico_sdk_init.cmake")
  message(FATAL_ERROR "Set PICO_SDK_PATH to pinned Pico SDK 2.3.0 (see README)")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(pico2_pc8001 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.3.0")
  message(FATAL_ERROR "This validation firmware pins Pico SDK 2.3.0")
endif()
add_executable(pico2_pc8001 main.c bridge.c usb_descriptors.c)
target_include_directories(pico2_pc8001 PRIVATE ${CMAKE_CURRENT_LIST_DIR})
pico_generate_pio_header(pico2_pc8001 ${CMAKE_CURRENT_LIST_DIR}/../pio/uart.pio)
target_link_libraries(pico2_pc8001 pico_stdlib pico_unique_id hardware_pio hardware_clocks hardware_watchdog tinyusb_device tinyusb_board)
target_compile_options(pico2_pc8001 PRIVATE -Wall -Wextra -Werror)
# Never let SDK stdio claim GP0/GP1 or mix console text into framed USB.
pico_enable_stdio_uart(pico2_pc8001 0)
pico_enable_stdio_usb(pico2_pc8001 0)
pico_set_program_name(pico2_pc8001 "Zig80 PC8001 Serial")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/bridge.h")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/bridge.h" bridge_version_line REGEX "^#define BRIDGE_VERSION ")
string(REGEX MATCH "\"([0-9]+\\.[0-9]+\\.[0-9]+)\"" bridge_version_match "${bridge_version_line}")
if(NOT bridge_version_match)
  message(FATAL_ERROR "BRIDGE_VERSION must contain a semantic version")
endif()
pico_set_program_version(pico2_pc8001 "${CMAKE_MATCH_1}")
pico_add_extra_outputs(pico2_pc8001)
